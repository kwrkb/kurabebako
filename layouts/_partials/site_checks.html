{{- /* サイト全体に掛ける検査。extend_head.html からトップページの描画時に 1 回だけ呼ばれる。 Hugo は date が未来の記事と expiryDate を過ぎた記事を、警告もエラーも出さずに ビルド対象から外す（LESSONS「記事の日付を『これから公開する時刻』で入れる」）。 ページが無いので extend_post_content.html の検査は走らず、ビルドログのページ数が 減ることでしか気づけない。そこで content/posts/ のファイル一覧と、実際に生成された ページ一覧を突き合わせ、draft でないのに生成されていない記事があれば止める。 `npm run dev` は --buildFuture 付きなので未来日付でも生成され、この検査は通る。 本番ビルド（hugo --gc --minify）で止まる */ -}} {{- /* 比較記事（posts）と作例（examples）の両方に掛ける */ -}} {{- range $section := slice "posts" "examples" -}} {{- $built := slice -}} {{- range where site.RegularPages "Section" $section -}} {{- with .File -}} {{- $built = $built | append (replace .Path "\\" "/") -}} {{- end -}} {{- end -}} {{- $dir := printf "content/%s" $section -}} {{- if fileExists $dir -}} {{- range os.ReadDir $dir -}} {{- if and (not .IsDir) (ne .Name "_index.md") (strings.HasSuffix .Name ".md") -}} {{- $path := printf "%s/%s" $section .Name -}} {{- $src := replaceRE "\r" "" (os.ReadFile (printf "content/%s" $path)) -}} {{- /* draft は front matter の中だけで見る。全文に掛けると本文のコード例の `draft = true` で 公開記事を draft と読み違える（2026-11-06 の監査 F01。push ゲートの isPublished と同じ修正） */ -}} {{- $fm := $src -}} {{- with findRESubmatch `(?s)^\+\+\+\n(.*?)\n\+\+\+` $src -}}{{- $fm = index (index . 0) 1 -}}{{- end -}} {{- $isDraft := findRE `(?m)^draft\s*=\s*true\s*(#.*)?$` $fm -}} {{- if and (not $isDraft) (not (in $built $path)) -}} {{- errorf "content/%s は draft ではないのにビルド対象に入っていません。date が未来か expiryDate を過ぎています。date を現在以前にしてください" $path -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
